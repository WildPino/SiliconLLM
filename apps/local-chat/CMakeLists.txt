cmake_minimum_required(VERSION 3.21)
project(SiliconLLMDesktop VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets Concurrent)

add_library(local_chat_core STATIC src/engine.cpp src/engine.h src/store.cpp src/store.h src/types.h)
target_link_libraries(local_chat_core PUBLIC Qt6::Core)
target_include_directories(local_chat_core PUBLIC src)
if(WIN32)
    target_link_libraries(local_chat_core PRIVATE psapi)
endif()
add_library(local_chat_widgets STATIC src/mainwindow.cpp src/mainwindow.h src/controller.cpp src/controller.h src/theme.cpp src/theme.h)
target_link_libraries(local_chat_widgets PUBLIC local_chat_core Qt6::Widgets Qt6::Concurrent)
if(WIN32)
    target_link_libraries(local_chat_widgets PRIVATE dwmapi)
endif()
add_executable(silicon-chat WIN32 src/main.cpp)
target_link_libraries(silicon-chat PRIVATE local_chat_widgets Qt6::Concurrent)

include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    foreach(suite store engine ui controller)
        add_executable(${suite}_tests ../../tests/local_chat/${suite}_tests.cpp)
        target_link_libraries(${suite}_tests PRIVATE local_chat_widgets Qt6::Test)
        add_test(NAME local_chat_${suite} COMMAND ${suite}_tests -o -,txt -o ${suite}-results.txt,txt)
        set_tests_properties(local_chat_${suite} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    endforeach()
    foreach(suite engine controller)
        target_compile_definitions(${suite}_tests PRIVATE
            LOCAL_CHAT_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/../../tests/local_chat/fixture_engine.py"
            LOCAL_CHAT_PYTHON="${Python3_EXECUTABLE}")
    endforeach()
endif()

install(TARGETS silicon-chat RUNTIME DESTINATION .)

option(LOCAL_CHAT_BUILD_UI_PROBE "Build the synthetic Windows UI cost probe" OFF)
if(LOCAL_CHAT_BUILD_UI_PROBE)
    add_executable(ui_cost_probe ../../tests/local_chat/ui_cost_probe.cpp)
    target_link_libraries(ui_cost_probe PRIVATE local_chat_widgets)
    target_compile_definitions(ui_cost_probe PRIVATE UI_COST_HAS_THEME=1)
    if(WIN32)
        target_link_libraries(ui_cost_probe PRIVATE psapi)
    endif()
endif()
